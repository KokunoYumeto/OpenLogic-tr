\olsection{\usetoken{P}{valuation} ve Sağlama}

\begin{defn}[!!^{valuation}s] 
$V$ bir doğruluk değerleri kümesi olsun. $\Lang{L}$ için $V$ değerli bir
\emph{!!{valuation}}, $V$'den !!a{element} seçip dilin her
!!{propositional variable} için atayan, yani
$\pAssign{v}\colon\PVar\to V$ olan bir fonksiyondur.
\end{defn}

\begin{defn}\ollabel{defn:pValue}
  Bir $\Log L$ çok değerli mantığının doğruluk değerleri kümesi $V$'ye giden
  !!a{valuation} $\pAssign{v}$ verildiğinde, değerlendirme fonksiyonu
  $\pValue{v}\colon\Frm[L]\to V$'yi tümevarımlı olarak şöyle tanımlayın:
  \begin{enumerate}
    \item $\pValue{v}(\Obj p_n) = \pAssign{v}(\Obj p_n)$; 
    \item $\star$ $0$-konumlu bir bağlaçsa
    $\pValue{v}(\star)=\tf{\star}[\Log L]$;
    \item $\star$ $n$-konumlu bir bağlaçsa
    \[
      \pValue{v}(\star(!A_1, \dots, !A_n)) = \tf{\star}[\Log L]
      (\pValue{v}(!A_1), \dots, \pValue{v}(!A_n)).
    \]
  \end{enumerate}
\end{defn}

\begin{defn}[Sağlama]
\ollabel{defn:satisfaction} !!^{formula} $!A$, !!a{valuation}
  $\pAssign{v}$ tarafından, $\pSat{v}{!A}[\Log L]$, ancak ve ancak
  $\pValue{v}(!A)[\Log L]\in V^+$ ise \emph{sağlanır}; burada $V^+$,
  $\Log L$'nin belirlenmiş doğruluk değerleri kümesidir.
  
  ``$\pSat{v}{!A}[\Log L]$ değil'' demek için
  $\pSat/{v}{!A}[\Log L]$ yazarız. $\Gamma$ bir !!{formula}s kümesiyse
  $\pSat{v}{\Gamma}[\Log L]$ ancak ve ancak her $!A\in\Gamma$ için
  $\pSat{v}{!A}[\Log L]$ ise.
\end{defn}
