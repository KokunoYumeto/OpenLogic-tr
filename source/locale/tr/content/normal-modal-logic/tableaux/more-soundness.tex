\olsection{Ek Kuralların Sağlamlığı}

!!a{tableau}{}daki bir dal, belirli bir sınıftaki modelde sağlanabilir olduğunda
kuralın uygulanmasıyla ortaya çıkan dal da o sınıftaki bir modelde
sağlanabiliyorsa, kurala bu modeller sınıfı için sağlam deriz.

\begin{prop}
  \ollabel{prop:soundness-T} 
  \iftag{prvBox}{\Ax{T}$\Box$}{}%
  \iftag{notprvBox,notprvDiamond}{}{ ve }%
  \iftag{prvDiamond}{\Ax{T}$\Diamond$}{}
  \iftag{notprvBox,notprvDiamond}{kuralı}{kuralları}
  yansımalı modeller için sağlamdır.
\end{prop}

\begin{proof}
\begin{tagenumerate}{prvBox,prvDiamond}
\tagitem{prvBox}{Dal, $\sFmla{\True}{\Box !B}[\sigma] \in \Gamma$'ya
  T$\Box$ uygulanarak genişletilsin:
  \iftag{probBox}{Alıştırma.}{Bunun sonucunda dala yeni bir !!{signed
      formula} $\sFmla{\True}{!B}[\sigma]$ eklenir.
    $\mSat{M}{\Gamma}[f]$, özellikle de $\mSat{M}{\Box
      !B}[f(\sigma)]$ olsun. $R$ yansımalı olduğundan
    $Rf(\sigma)f(\sigma)$. Dolayısıyla $\mSat{M}{!B}[f(\sigma)]$;
    yani $\mModel{M},f$, $\sFmla{\True}{!B}[\sigma]$'yı sağlar.}}{}
\tagitem{prvDiamond}{Dal, $\sFmla{\False}{\Diamond !B}[\sigma] \in
  \Gamma$'ya T$\Diamond$ uygulanarak genişletilsin:
  \iftag{probDiamond}{Alıştırma.}{Bunun sonucunda dala yeni bir !!{signed
      formula} $\sFmla{\False}{!B}[\sigma]$ eklenir.
    $\mSat{M}{\Gamma}[f]$, özellikle de $\mSat/{M}{\Diamond
      !B}[f(\sigma)]$ olsun. $R$ yansımalı olduğundan
    $Rf(\sigma)f(\sigma)$. Dolayısıyla $\mSat/{M}{!B}[f(\sigma)]$;
    yani $\mModel{M},f$, $\sFmla{\False}{!B}[\sigma]$'yı sağlar.}}{}
\end{tagenumerate}
\end{proof}

\tagprob{probBox,probDiamond}
\begin{prob}
  \olref[nml][tab][msn]{prop:soundness-T} önermesinin ispatını tamamlayın.
\end{prob}
\tagendprob

    $w\in W$ vardır. O zaman $\mSat/{M}{!B}[w]$, dolayısıyla
    $\mSat/{M}{\Box !B}[f(\sigma)]$. Böylece $\mModel{M},f$,

\begin{prop}
  \ollabel{prop:soundness-B} 
  \iftag{prvBox}{\Ax{B}$\Box$}{}%
  \iftag{notprvBox,notprvDiamond}{}{ ve }%
  \iftag{prvDiamond}{\Ax{B}$\Diamond$}{}
  \iftag{notprvBox,notprvDiamond}{kuralı}{kuralları}
  simetrik modeller için sağlamdır.
\end{prop}

\begin{proof}
\begin{tagenumerate}{prvBox,prvDiamond}
\tagitem{prvBox}{Dal, $\sFmla{\True}{\Box !B}[\sigma.n] \in \Gamma$'ya
  B$\Box$ uygulanarak genişletilsin:
  \iftag{probBox}{Alıştırma.}{Bunun sonucunda dala yeni bir !!{signed
      formula} $\sFmla{\True}{!B}[\sigma]$ eklenir.
    $\mSat{M}{\Gamma}[f]$, özellikle de $\mSat{M}{\Box
      !B}[f(\sigma.n)]$ olsun. $f$, daldaki öneklerin $\mModel{M}$ içindeki
    bir yorumu olduğundan $Rf(\sigma)f(\sigma.n)$. $R$ simetrik olduğundan
    $Rf(\sigma.n)f(\sigma)$. $\mSat{M}{\Box !B}[f(\sigma.n)]$ olduğundan
    $\mSat{M}{!B}[f(\sigma)]$. Dolayısıyla $\mModel{M},f$,
    $\sFmla{\True}{!B}[\sigma]$'yı sağlar.}}{}
\tagitem{prvDiamond}{Dal, $\sFmla{\False}{\Diamond !B}[\sigma.n] \in
  \Gamma$'ya B$\Diamond$ uygulanarak genişletilsin:
  \iftag{probDiamond}{Alıştırma.}{Bunun sonucunda dala yeni bir !!{signed
      formula} $\sFmla{\False}{!B}[\sigma]$ eklenir.
    $\mSat{M}{\Gamma}[f]$, özellikle de $\mSat/{M}{\Diamond
      !B}[f(\sigma.n)]$ olsun. $f$, daldaki öneklerin $\mModel{M}$ içindeki
    bir yorumu olduğundan $Rf(\sigma)f(\sigma.n)$. $R$ simetrik olduğundan
    $Rf(\sigma.n)f(\sigma)$. $\mSat/{M}{\Diamond !B}[f(\sigma.n)]$
    olduğundan $\mSat/{M}{!B}[f(\sigma)]$. Dolayısıyla $\mModel{M},f$,
    $\sFmla{\False}{!B}[\sigma]$'yı sağlar.}}{}
\end{tagenumerate}
\end{proof}

\tagprob{probBox,probDiamond}
\begin{prob}
  \olref[nml][tab][msn]{prop:soundness-B} önermesinin ispatını tamamlayın.
\end{prob}
\tagendprob

  \iftag{notprvBox,notprvDiamond}{kuralı}{kuralları}
  geçişli modeller için sağlamdır.

\begin{proof}
\begin{tagenumerate}{prvBox,prvDiamond}

    $Rf(\sigma)f(\sigma.n)$. Şimdi $Rf(\sigma.n)w$ olan gelişigüzel bir $w$
    dünyası alalım. $R$ geçişli olduğundan $Rf(\sigma)w$.
    $\mSat{M}{\Box !B}[f(\sigma)]$ olduğundan $\mSat{M}{!B}[w]$.

    $\sFmla{\True}{\Box !B}[\sigma.n]$'yı sağlar.}}{}
