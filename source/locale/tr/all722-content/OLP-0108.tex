\olsection{\usetoken{S}{derivability} ve Niceleyiciler}

\begin{explain}
  Tamlık teoremi ayrıca tableau kurallarının,~$\Proves$ hakkında bu bölümde
  ortaya konan olguları vermesini gerektirir.
\end{explain}

\begin{thm}
\ollabel{thm:strong-generalization} $c$, $\Gamma$ veya~$!A(x)$ içinde
geçmeyen bir sabitse ve $\Gamma \Proves !A(c)$ ise
$\Gamma \Proves \lforall[x][!A(x)]$.
\end{thm}

\begin{proof}
$\Gamma \Proves !A(c)$ olduğunu varsayın; yani $!B_1$, \dots,
$!B_n \in \Gamma$ ve aşağıdaki küme için kapalı bir !!{tableau} vardır:
\begin{align*}
\{ \sFmla{\False}{!A(c)}, &
\sFmla{\True}{!B_1}, \dots, \sFmla{\True}{!B_n} \}.
\intertext{Şu küme için de kapalı bir !!{tableau} bulunduğunu göstermeliyiz:}
\{ \sFmla{\False}{\lforall[x][!A(x)]}, &
\sFmla{\True}{!B_1}, \dots, \sFmla{\True}{!B_n} \}.
\end{align*}
Kapalı !!{tableau}'yu alın, ilk varsayımı
$\sFmla{\False}{\lforall[x][!A(x)]}$ ile değiştirin ve varsayımlardan sonra
$\sFmla{\False}{!A(c)}$ ifadesini ekleyin.
\begin{center}
\begin{tableau}{not line numbering}
  [\sFmla{\False}{\formula{A}(c)}
    [\sFmla{\True}{\formula{B}_1}
      [\vdots
      [\sFmla{\True}{\formula{B}_n}, 
        [][][]]]]]
\end{tableau}\qquad
\begin{tableau}{not line numbering}
  [\sFmla{\False}{\lforall[x][\formula{A}(x)]}
    [\sFmla{\True}{\formula{B}_1}
      [\vdots
        [\sFmla{\True}{\formula{B}_n},
          [\sFmla{\False}{\formula{A}(c)}
        [][][]]]]]]
\end{tableau}
\end{center}
Daha önce varsayım olarak bulunan bütün !!{sentence}s, !!{tableau}'nun
üstünde hâlâ bulunduğundan tableau yine kapalıdır. Eklenen satır,
$\TRule{\False}{\lforall}$ kuralının doğru uygulanmasının sonucudur; çünkü
!!{constant}~$c$, $!B_1$, \dots, $!B_n$ veya $\lforall[x][!A(x)]$ içinde,
yani yeni !!{tableau}'da eklenen satırın üstünde geçmez.
\end{proof}

\begin{prop}
\ollabel{prop:provability-quantifiers}
\begin{tagenumerate}{prvEx,prvAll}
\tagitem{prvEx}{$!A(t) \Proves \lexists[x][!A(x)]$.}{}
\tagitem{prvAll}{$\lforall[x][!A(x)] \Proves !A(t)$.}{}
\end{tagenumerate}
\end{prop}

\begin{proof}
\begin{tagenumerate}{prvEx,prvAll}
\tagitem{prvEx}{$\sFmla{\False}{\lexists[x][!A(x)]},
  \sFmla{\True}{!A(t)}$ için kapalı bir !!{tableau} şöyledir:
  \begin{oltableau}
    [\sFmla{\False}{\lexists[x][\formula{A}(x)]}, just = \TAss
      [\sFmla{\True}{\formula{A}(t)}, just = \TAss
        [\sFmla{\False}{\formula{A}(t)}, just = {\TRule{\False}{\lexists}[1]}, close]]]
    \end{oltableau}}{}
\tagitem{prvAll}{$\sFmla{\False}{\formula{A}(t)},
  \sFmla{\True}{\lforall[x][!A(x)]}$ için kapalı bir !!{tableau} şöyledir:
  \begin{oltableau}
    [\sFmla{\False}{\formula{A}(t)}, just = \TAss
      [\sFmla{\True}{\lforall[x][\formula{A}(x)]}, just = \TAss
        [\sFmla{\True}{\formula{A}(t)}, just = {\TRule{\True}{\lforall}[2]}, close]]]
    \end{oltableau}}{}
\end{tagenumerate}
\end{proof}
