\olsection{Birinci Dereceden Diller İçin Örtülmüş
\printtoken{P}{structure}}

\begin{explain}
Bir terimin hiç !!{variable}s içermiyorsa \emph{kapalı} olduğunu
hatırlayın.
\end{explain}

\begin{defn}[Kapalı terimlerin !!^{value}]
$t$,~$\Lang L$ dilinin kapalı bir terimi ve $\Struct M$,~$\Lang L$
için !!{structure} ise \emph{!!{value}}~$\Value tM$ aşağıdaki gibi
tanımlanır:
\begin{enumerate}
\item $t$ yalnızca !!{constant}~$c$ ise $\Value cM=\Assign cM$.
\item $t$,~$\Atom f{t_1,\ldots,t_n}$ biçimindeyse
  \[
  \Value tM=\Assign fM(\Value{t_1}M,\ldots,\Value{t_n}M).
  \]
\end{enumerate}
\end{defn}

\begin{defn}[Örtülmüş !!{structure}]
Bir !!{structure}, evrenindeki her eleman kapalı bir terimin
!!{value} ise \emph{örtülmüş}tür.
\end{defn}

\begin{ex}
$\Lang L$; !!{constant}s~$\Obj{zero}$, $\Obj{one}$, $\Obj{two}$,
\dots; iki-li !!{predicate}~$<$; ve iki-li !!{function}s~$+$ ile
$\times$ simgelerini içeren dil olsun. O zaman~$\Lang L$ için
!!a{structure}~$\Struct M$, evreni
$\Domain M=\{0,1,2,\ldots\}$ ve atamaları
$\Assign{\Obj{zero}}M=0$, $\Assign{\Obj{one}}M=1$,
$\Assign{\Obj{two}}M=2$ vb. olan yapıdır. İki-li bağıntı simgesi~$<$
için $\Assign< M$ kümesi, $c_1$'in~$c_2$'den küçük olduğu bütün
$\tuple{c_1,c_2}\in\Domain M^2$ ikililerinin kümesidir. Örneğin
$\tuple{1,3}\in\Assign< M$, fakat $\tuple{2,2}\notin\Assign< M$.
İki-li !!{function}~$+$ için $\Assign+M$'yi olağan biçimde
tanımlayalım; örneğin $\Assign+M(2,3)$ değeri~$5$'tir. İki-li
!!{function}~$\times$ için de benzer biçimde tanımlayalım. Böylece
$\Obj{four}$ simgesinin !!{value} tam olarak~$4$; ve
$\times(\Obj{two},+(\Obj{three},\Obj{zero}))$ ifadesinin (ya da
araeklemli gösterimle
$\Obj{two}\times(\Obj{three}+\Obj{zero})$) !!{value} şudur:
\begin{multline*}
\Value{\times(\Obj{two}, +(\Obj{three},\Obj{zero}))}{M} =\\
\begin{aligned}
& =\Assign{\times}{M}(\Value{\Obj{two}}{M}, \Value{+(\Obj{three}, \Obj{zero})}{M})\\
& = \Assign{\times}{M}(\Value{\Obj{two}}{M}, \Assign{+}{M}(\Value{\Obj{three}}{M},
\Value{\Obj{zero}}{M})) \\
& = \Assign{\times}{M}(\Assign{\Obj{two}}{M}, \Assign{+}{M}(\Assign{\Obj{three}}{M},
\Assign{\Obj{zero}}{M})) \\
& = \Assign{\times}{M}(2, \Assign{+}{M}(3, 0)) \\
& = \Assign{\times}{M}(2, 3) \\
& = 6.
\end{aligned}
\end{multline*}
\end{ex}
